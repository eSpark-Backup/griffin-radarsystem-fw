57 /**
58  * @brief Function for creating envelope service

60 static acc_service_handle_t create_envelope_service(void)
61 {
62  bool success = true;
63  acc_service_handle_t service_handle = NULL;
64 
65  float range_start_m = 0.2f;
66  float range_length_m = RANGE_LENGTH;
67 
68  acc_service_configuration_t envelope_configuration = acc_service_envelope_configuration_create();
69 
70  if (envelope_configuration == NULL)
71  {
72  printf("Could not create envelope configuration\n");
73  success = false;
74  }
75 
76  if (success)
77  {
78  acc_service_requested_start_set(envelope_configuration, range_start_m);
79  acc_service_requested_length_set(envelope_configuration, range_length_m);
80  acc_service_power_save_mode_set(envelope_configuration, SELECTED_POWER_SAVE_MODE);
81  acc_service_profile_set(envelope_configuration, SELECTED_SERVICE_PROILE);
82  acc_service_hw_accelerated_average_samples_set(envelope_configuration, HWAAS);
83  acc_service_envelope_downsampling_factor_set(envelope_configuration, DOWNSAMPLING_FACTOR);
84 
85  service_handle = acc_service_create(envelope_configuration);
86 
87  if (service_handle == NULL)
88  {
89  printf("Could not create envelope service\n");
90  success = false;
91  }
92  }
93 
94  acc_service_envelope_configuration_destroy(&envelope_configuration);
95 
96  return success ? service_handle : NULL;

100 /**
101  * @brief Function for running the envelope service
102  */
103 static void execute_envelope_service(acc_service_handle_t handle, uint16_t *result1, uint16_t *result2)
104 {
105  acc_service_envelope_metadata_t metadata = { 0 };
106 
107  acc_service_envelope_get_metadata(handle, &metadata);
108 
109  uint16_t envelope_data[metadata.data_length];
110  acc_service_envelope_result_info_t result_info;
111 
112  uint16_t max_amplitude_index = 0;
113  uint16_t max_amplitude = 0;
114 
115  if (acc_service_envelope_get_next(handle, envelope_data, metadata.data_length, &result_info))
116  {
117  max_amplitude_index = 0;
118  max_amplitude = 0;
119 
120  for (uint16_t index = 0; index < metadata.data_length; index++)
121  {
122  if (envelope_data[index] > max_amplitude)
123  {
124  max_amplitude = envelope_data[index];
125  max_amplitude_index = index;
126  }
127  }
128  }
129 
130  *result1 = max_amplitude;
131  *result2 = max_amplitude_index;
132 }

138 int acconeer_main(int argc, char *argv[])
139 {
140  (void)argc;
141  (void)argv;
142  printf("Acconeer software version %s\n", acc_version_get());
143 
144  acc_hal_t hal = *acc_hal_integration_get_implementation();
145 
146  acc_integration_set_periodic_wakeup(SUSPEND_TIME_BETWEEN_UPDATES_MS);
147 
148  hal.log.log_level = ACC_LOG_LEVEL_ERROR;
149 
150  if (!acc_rss_activate(&hal))
151  {
152  return EXIT_FAILURE;
153  }
154 
155  acc_service_handle_t handle = create_envelope_service();
156  if (handle == NULL)
157  {
158  printf("acc_service_create() failed\n");
159  return EXIT_FAILURE;
160  }
161 
162  if (!acc_service_activate(handle))
163  {
164  printf("acc_service_activate() failed\n");
165  acc_service_destroy(&handle);
166  acc_rss_deactivate();
167  return EXIT_FAILURE;
168  }
169 
170  while (true)
171  {
172  uint16_t result[2] = {0};
173 
174  execute_envelope_service(handle, &result[0], &result[1]);
175 
176  printf("Service data %" PRIu16 " , %" PRIu16 "\n", result[0], result[1]);
177 
178  acc_integration_sleep_until_periodic_wakeup();
179  }
180 
181  acc_service_deactivate(handle);
182  acc_service_destroy(&handle);
183  acc_rss_deactivate();
184 
185  return EXIT_SUCCESS;
186 }
